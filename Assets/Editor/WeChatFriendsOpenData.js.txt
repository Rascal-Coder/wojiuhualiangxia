/* Friend data stays in this open-data context. No demo or generated scores. */
'use strict';
const KEY = 'inkomino_completed_v1';
const VERSION = 'friends-20261103-r10';
const canvas = wx.getSharedCanvas(), ctx = canvas.getContext('2d');
const PAGE_SIZE = 5, CACHE_TTL = 30000, REFRESH_COOLDOWN = 1500;
let visible = false, requestedFriends = false, rows = [], page = 0, state = 'idle';
let inFlight = null, hasSnapshot = false, lastSuccess = 0, lastAttempt = -Infinity;
let dataVersion = 0, snapshotVersion = -1, lastAttemptVersion = -1;
let pendingScore = -1, knownScore = -1, syncing = false, syncNotice = '';
let retryTimer = null, retryCount = 0, failureCode = '', refreshNotice = '';
let animationTimer = null, repaint = null;
const avatars = Object.create(null);
const C = {ink:'#493362', muted:'#8a739f', grape:'#6645aa', coral:'#ef628b', paper:'#fff9ee', edge:'#e7d8f2'};
const gameType = require('./font')(wx, ctx, () => draw());
const gameArt = require('./art')(wx, ctx, () => draw());
console.log('[INK_FRIENDS] init ' + VERSION);

function score(value) {
  if (typeof value !== 'string' && typeof value !== 'number') return 0;
  const n = Number(value);
  return Number.isFinite(n) ? Math.max(0, Math.min(500, Math.floor(n))) : 0;
}
function valueOf(kv) {
  const entry = (Array.isArray(kv) ? kv : []).find(item => item && item.key === KEY);
  return score(entry ? entry.value : 0);
}
function queueScore(value) {
  pendingScore = Math.max(pendingScore, score(value)); retryCount = 0; syncScore();
}
function syncFailure() {
  syncing = false; syncNotice = '进度同步稍后重试'; draw();
  if (!retryTimer && retryCount < 3) {
    retryCount++;
    retryTimer = setTimeout(() => { retryTimer = null; syncScore(); }, retryCount * 15000);
  }
}
function syncScore() {
  if (syncing || pendingScore < 0 || pendingScore <= knownScore) return;
  syncing = true;
  let finished = false;
  const timer = setTimeout(() => { if (!finished) { finished = true; syncFailure(); } }, 12000);
  function failure() { if (finished) return; finished = true; clearTimeout(timer); syncFailure(); }
  function finish(value, written = false) {
    if (finished) return;
    finished = true; clearTimeout(timer); knownScore = Math.max(knownScore, value);
    syncing = false; syncNotice = ''; retryCount = 0;
    if (retryTimer !== null) { clearTimeout(retryTimer); retryTimer = null; }
    // Only committed writes invalidate the snapshot. Concurrent commits are
    // coalesced into one follow-up read, never a fabricated local rank.
    if (written) dataVersion++;
    if (pendingScore > knownScore) syncScore();
    if (written && requestedFriends && visible) fetchFriends(); else draw();
  }
  try {
    wx.getUserCloudStorage({keyList: [KEY], success(result) {
      if (finished) return;
      const best = Math.max(valueOf(result.KVDataList), pendingScore, knownScore);
      if (best === valueOf(result.KVDataList)) { finish(best); return; }
      try {
        wx.setUserCloudStorage({KVDataList: [{key: KEY, value: String(best)}],
          success() { finish(best, true); }, fail: failure});
      } catch (_) { failure(); }
    }, fail: failure});
  } catch (_) { failure(); }
}
function fetchFriends(force = false) {
  if (inFlight) { draw(); return; }
  const now = Date.now(), dirty = dataVersion !== snapshotVersion;
  if (!force && hasSnapshot && !dirty && now - lastSuccess < CACHE_TTL) { draw(); return; }
  // Share cooldown across taps/focus events; a newly committed score bypasses it.
  if (now - lastAttempt < REFRESH_COOLDOWN && dataVersion <= lastAttemptVersion) { draw(); return; }
  const task = {version: dataVersion}; inFlight = task;
  lastAttempt = now; lastAttemptVersion = dataVersion;
  if (!hasSnapshot) state = 'loading';
  failureCode = ''; refreshNotice = '';
  const timeout = setTimeout(() => failed({errMsg:'getFriendCloudStorage timeout'}, 'TIMEOUT'), 12000);
  draw(); console.log('[INK_FRIENDS] fetch start');
  function done() {
    if (inFlight !== task) return false;
    inFlight = null; clearTimeout(timeout); return true;
  }
  function failed(error, fallback = 'API_FAILURE') {
    if (!done()) return;
    const message = String(error && (error.errMsg || error.message) || 'unknown error');
    const raw = error && (error.errno ?? error.errCode ?? error.errcode);
    failureCode = /^-?\d{1,9}$/.test(String(raw)) ? 'WX_' + raw : fallback;
    console.warn('[INK_FRIENDS] fetch failed: ' + message.slice(0,180) + ' code=' + failureCode);
    const configuration = [1026,112,101100,101101].includes(Number(raw)) || /privacy api banned|please go to mp.*(?:privacy|official popup)|onNeedPrivacyAuthorization.*announce/i.test(message);
    const permission = [103,104,101102].includes(Number(raw)) || /auth deny|auth denied|authorize:fail|permission denied|scope\.WxFriendInteraction|privacy permission is not authorized|user deny|user denied/i.test(message);
    if (configuration || permission) {
      // Revoked/configuration-blocked friend data must not survive in memory.
      rows = []; page = 0; hasSnapshot = false; snapshotVersion = -1;
      Object.keys(avatars).forEach(url => delete avatars[url]);
      gameType.clear();
      state = configuration ? 'configuration' : 'permission';
    } else if (hasSnapshot) refreshNotice = '更新未完成，点击刷新';
    else state = 'error';
    draw();
    // A read started before a score commit is obsolete even when it fails.
    // Retry once for that newer version, but never loop on permission errors.
    if (!configuration && !permission && dataVersion !== task.version && requestedFriends && visible) fetchFriends();
  }
  try {
    if (typeof wx.getFriendCloudStorage !== 'function') { failed({errMsg:'getFriendCloudStorage unavailable'}, 'API_UNAVAILABLE'); return; }
    wx.getFriendCloudStorage({keyList: [KEY], success(result) {
      if (!done()) return;
      const seen = Object.create(null);
      rows = (result && Array.isArray(result.data) ? result.data : []).filter(item => {
        if (!item || !Array.isArray(item.KVDataList) || !item.KVDataList.some(k => k && k.key === KEY)) return false;
        if (item.openid && seen[item.openid]) return false;
        if (item.openid) seen[item.openid] = true;
        return true;
      }).map(item => ({name:String(item.nickname || '微信好友'), avatar:String(item.avatarUrl || ''), score:valueOf(item.KVDataList)}));
      rows.sort((a,b) => b.score-a.score || a.name.localeCompare(b.name));
      hasSnapshot = true; lastSuccess = Date.now(); snapshotVersion = task.version;
      state = rows.length ? 'ready' : 'empty'; page = Math.min(page, maxPage());
      const urls = new Set(rows.map(item => item.avatar));
      Object.keys(avatars).forEach(url => { if (!urls.has(url) || avatars[url].failed) delete avatars[url]; });
      draw(); console.log('[INK_FRIENDS] fetch success count=' + rows.length);
      if (dataVersion !== task.version && requestedFriends && visible) fetchFriends();
    }, fail: failed});
  } catch (error) { failed(error, 'CALL_ERROR'); }
}
function maxPage() { return Math.max(0, Math.ceil(rows.length / PAGE_SIZE) - 1); }
function text(value,x,y,size,color=C.ink,align='center',role='body') {
  gameType.paint(value,x,y,size,color,align,role);
}
function shortName(value,width,size) {
  return gameType.truncate(value,width,size);
}
function roundPath(x,y,w,h,r) {
  r=Math.max(0,Math.min(r,w/2,h/2));ctx.beginPath();ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();
}
function box(x,y,w,h,r,fill,stroke=null,line=1) {
  roundPath(x,y,w,h,r);ctx.fillStyle=fill;ctx.fill();
  if(stroke){ctx.strokeStyle=stroke;ctx.lineWidth=line;ctx.stroke();}
}
function circle(x,y,r,fill) {ctx.beginPath();ctx.arc(x,y,r,0,Math.PI*2);ctx.fillStyle=fill;ctx.fill();}
function avatarImage(url) {
  if(!url||!/^https:\/\//i.test(url))return null;
  const cached=avatars[url];
  if(cached){cached.used=Date.now();return cached.ready?cached.image:null;}
  // Only load visible-page avatars. LRU avoids the old permanent image cutoff.
  const urls=Object.keys(avatars);
  if(urls.length>=40){urls.sort((a,b)=>avatars[a].used-avatars[b].used);delete avatars[urls[0]];}
  let image;try{image=wx.createImage();}catch(_){return null;}
  const entry={image,ready:false,failed:false,used:Date.now()};avatars[url]=entry;
  image.onload=()=>{if(avatars[url]!==entry)return;entry.ready=true;draw();};
  image.onerror=()=>{entry.failed=true;};image.src=url;return null;
}
function avatar(item,x,y,size) {
  const image=avatarImage(item.avatar);
  box(x,y,size,size,size*.28,'#eadff9','#ffffff',3*size/80);
  if(image){
    ctx.save();roundPath(x+2,y+2,size-4,size-4,size*.25);ctx.clip();
    try{const iw=image.width||size,ih=image.height||size,side=Math.min(iw,ih);
      ctx.drawImage(image,(iw-side)/2,(ih-side)/2,side,side,x,y,size,size);}catch(_){}
    ctx.restore();
  }else{
    circle(x+size*.5,y+size*.38,size*.16,'#b6a0d5');
    box(x+size*.22,y+size*.58,size*.56,size*.29,size*.14,'#b6a0d5');
  }
}
function medal(rank,x,y,u,rowHeight) {
  if(rank>3){text(String(rank),x,y,44*u,C.muted);return;}
  const anchor=gameArt.paint('rank'+rank,x,y,88*u,rowHeight-8*u);
  text(String(rank),anchor?anchor.x:x,anchor?anchor.y:y,42*u,C.ink);
}
function stopAnimation() {
  if(animationTimer!==null){clearTimeout(animationTimer);animationTimer=null;}
  if(repaint!==null){cancelAnimationFrame(repaint);repaint=null;}
}
function surfaceFrame() {
  draw();
  // ShowOpenData clears the native canvas even at the same size. Draw now and
  // once next frame; the SDK copies textures itself, without a static RAF loop.
  if(repaint===null&&visible)repaint=requestAnimationFrame(()=>{repaint=null;draw();});
}
function loadingDot(x,y,u) {
  const phase=Math.floor(Date.now()/180)%3;
  for(let i=0;i<3;i++)circle(x+i*17*u,y,5*u,i===phase?C.grape:'#d8c7ee');
}
function draw() {
  if(!visible||canvas.width<1||canvas.height<1)return;
  if(animationTimer!==null){clearTimeout(animationTimer);animationTimer=null;}
  const w=canvas.width,h=canvas.height,u=w/960;
  ctx.clearRect(0,0,w,h);ctx.fillStyle=C.paper;ctx.fillRect(0,0,w,h);
  if(state==='ready'||state==='loading'){
    text('名次',66*u,37*u,42*u,C.muted);text('好友',220*u,37*u,42*u,C.muted,'left');
    text('已通关',w-48*u,37*u,42*u,C.muted,'right');
    const top=82*u,gap=12*u,rowHeight=Math.min(132*u,Math.max(84*u,(h-top-78*u-gap*4)/PAGE_SIZE));
    const pageRows=rows.slice(page*PAGE_SIZE,(page+1)*PAGE_SIZE),count=state==='loading'?PAGE_SIZE:pageRows.length;
    for(let i=0;i<count;i++){
      const y=top+i*(rowHeight+gap),cy=y+rowHeight/2,rank=page*PAGE_SIZE+i+1;
      const tint=state==='loading'?'#fffdf7':rank===1?'#fff7dc':rank===2?'#f0faf4':rank===3?'#f6efff':'#fffcf6';
      box(15*u,y+4*u,w-30*u,rowHeight,23*u,'#ebddf5');box(15*u,y,w-30*u,rowHeight,23*u,tint,C.edge,2*u);
      const size=Math.min(82*u,rowHeight-18*u),ax=119*u,ay=cy-size/2;
      if(state==='loading'){
        const alpha=.30+(.5+.5*Math.sin(Date.now()/420+i))*.16,fill='rgba(174,150,207,'+alpha+')';
        circle(66*u,cy,22*u,fill);
        box(ax,ay,size,size,size*.28,fill);
        box(220*u,cy-14*u,(i%2?236:304)*u,28*u,14*u,fill);
        box(w-213*u,cy-32*u,177*u,64*u,25*u,fill);
      }else{
        const item=pageRows[i];medal(rank,66*u,cy,u,rowHeight);avatar(item,ax,ay,size);
        text(shortName(item.name,w-465*u,46*u),220*u,cy,46*u,C.ink,'left');
        box(w-213*u,cy-32*u,177*u,64*u,25*u,rank===1?'#ffedb5':'#f0e6fb');text(item.score+' 关',w-124*u,cy,44*u,C.grape);
      }
    }
    const footY=h-29*u;
    if(inFlight){loadingDot(32*u,footY,u);text(hasSnapshot?'正在更新…':'正在读取好友进度…',84*u,footY,42*u,C.muted,'left');}
    else text(refreshNotice||syncNotice||('共 '+rows.length+' 位好友'),28*u,footY,42*u,refreshNotice?C.coral:C.muted,'left');
    if(state==='ready')text((page+1)+' / '+(maxPage()+1)+' 页',w-28*u,footY,42*u,C.muted,'right');
  }else{
    gameArt.paint('empty',w/2,h*.28,340*u,h*.42);
    const messages={
      idle:['和好友一起闯关','点击刷新查看好友排行'],
      empty:['还没有同玩好友的成绩',''],
      permission:['好友互动尚未授权','允许好友互动后，点击下方刷新'],
      configuration:['好友排行暂不可用','隐私功能配置尚未就绪'],
      error:['好友排行加载未完成','请检查网络后，点击下方刷新']
    };
    const lines=messages[state]||messages.error;
    text(lines[0],w/2,h*.53,47*u,C.ink,'center','display');text(lines[1],w/2,h*.66,42*u,C.muted);
    if(failureCode)text('错误码：'+failureCode,w/2,h*.79,42*u,C.muted);
    if(inFlight){loadingDot(w/2-118*u,h-32*u,u);text('正在更新…',w/2+35*u,h-32*u,42*u,C.muted);}
    else if(refreshNotice||syncNotice)text(refreshNotice||syncNotice,w/2,h-32*u,42*u,C.muted);
  }
  if(inFlight)animationTimer=setTimeout(()=>{animationTimer=null;draw();},180);
}
wx.onMessage(message=>{
  if(typeof message==='string'){
    try{message=JSON.parse(message);}catch(_){console.warn('[INK_FRIENDS] ignored invalid JSON message');return;}
  }
  if(!message)return;
  switch(message.type){
    case 'score:update':queueScore(message.score);break;
    case 'WXRender':case 'WXShow':
      if(message.type==='WXShow'&&!requestedFriends)break;
      if(message.width>0&&canvas.width!==message.width)canvas.width=message.width;
      if(message.height>0&&canvas.height!==message.height)canvas.height=message.height;
      visible=true;if(requestedFriends)fetchFriends();surfaceFrame();break;
    case 'WXDestroy':visible=false;stopAnimation();break;
    case 'friends:hide':visible=false;requestedFriends=false;stopAnimation();break;
    case 'friends:prepare':gameType.prepare();gameArt.prepare();break;
    case 'friends:show':requestedFriends=true;visible=true;fetchFriends();surfaceFrame();break;
    case 'friends:resume':if(requestedFriends&&visible){syncScore();fetchFriends();}break;
    case 'friends:refresh':if(requestedFriends&&visible){syncScore();fetchFriends(true);}break;
    case 'friends:page':page=Math.max(0,Math.min(maxPage(),page+(message.delta<0?-1:1)));draw();break;
  }
});
